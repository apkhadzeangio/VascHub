async function createVariableCollection(name, modeNames) {
  const existing=(await figma.variables.getLocalVariableCollectionsAsync()).find(c=>c.name===name);
  const collection=existing||figma.variables.createVariableCollection(name);
  collection.renameMode(collection.modes[0].modeId,modeNames[0]);
  return {collection,modeIds:{[modeNames[0]]:collection.modes[0].modeId}};
}

function token(name,type,value,scopes,css){
  let v=existingVars.find(v=>v.variableCollectionId===collection.id&&v.name===name);
  if(!v) v=figma.variables.createVariable(name,collection,type);
  v.scopes=scopes;v.setValueForMode(modeIds.Light,value);v.setVariableCodeSyntax('WEB',`var(${css})`);variables[name]=v;return v;
}

function paint(name){return figma.variables.setBoundVariableForPaint({type:'SOLID',color:{r:0,g:0,b:0}},'color',variables['color/'+name]);}
function fill(n,name){n.fills=name?[paint(name)]:[];}
function stroke(n,name='border'){n.strokes=[paint(name)];n.strokeWeight=1;n.setBoundVariable('strokeWeight',variables['border/width']);}

function icon(parent,name,size=22,color='primary'){
  const source=ICONS[name];if(!source)throw new Error('Missing source SVG '+name);
  const n=track(figma.createNodeFromSvg(source.replaceAll('currentColor',colors[color])));n.name='Icon/'+name;n.resize(size,size);parent.appendChild(n);

  return n;
}
